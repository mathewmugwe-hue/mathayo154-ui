async function processMatches() { const outputEl = document.getElementById('output'); const text = document.getElementById('matchData').value.trim(); if (!text) { outputEl.innerHTML = 'Please provide match text or upload a screenshot first.'; return; } outputEl.innerHTML = 'Parsing entries and executing v3.4 backend pipeline...'; const lines = text.split('\n').map(l => l.trim()).filter(Boolean); let parsedMatches = []; let i = 0; while (i < lines.length) { let line = lines[i]; // Skip headers or metadata if (line.includes("Soccer -") || line.includes("ID:") || line.includes("Market") || line === "1" || line === "X" || line === "2") { i++; continue; } // Format A: Comma separated (TeamA, TeamB, HomeOdds, DrawOdds, AwayOdds) if (line.includes(',')) { let parts = line.split(',').map(s => s.trim()); if (parts.length >= 5) { parsedMatches.push({ home: parts[0].replace(/\s+[Cc]$/, '').trim(), away: parts[1].replace(/\s+[Cc]$/, '').trim(), h: parts[2], d: parts[3], a: parts[4] }); } i++; continue; } // Format B: Inline "TeamA vs TeamB" if (line.toLowerCase().includes(' vs ')) { let teams = line.split(/vs/i).map(s => s.trim()); if (teams.length >= 2) { parsedMatches.push({ home: teams[0].replace(/\s+[Cc]$/, '').trim(), away: teams[1].replace(/\s+[Cc]$/, '').trim(), h: "2.20", d: "3.40", a: "3.10" }); } i++; continue; } // Format C: Vertical Sportsbook Stack (Team 1 on line i, Team 2 on line i+1, followed by odds in next lines) if (i + 1 < lines.length) { let potentialHome = lines[i]; let potentialAway = lines[i+1]; // Ensure these aren't random numbers or generic labels if (isNaN(potentialHome) && isNaN(potentialAway) && !potentialHome.includes("Soccer") && !potentialAway.includes("Soccer")) { // Scan next 15 lines for the first 3 valid float odds values let oddsCandidates = []; for (let j = i + 2; j < i + 18 && j < lines.length; j++) { let val = parseFloat(lines[j]); if (!isNaN(val) && val >= 1.01 && val < 50.0) { oddsCandidates.push(lines[j]); } } if (oddsCandidates.length >= 3) { parsedMatches.push({ home: potentialHome.replace(/\s+[Cc]$/, '').trim(), away: potentialAway.replace(/\s+[Cc]$/, '').trim(), h: oddsCandidates[0], d: oddsCandidates[1], a: oddsCandidates[2] }); i += 2; // Move past team lines; inner loop handles odds skipping continue; } } } i++; } if (parsedMatches.length === 0) { outputEl.innerHTML = `Could not isolate match items automatically. Please ensure fixtures contain team names and odds.`; return; } let html = ''; for (let m of parsedMatches) { try { const url = `https://mathayo-backend.onrender.com/predict?home_team=${encodeURIComponent(m.home)}&away_team=${encodeURIComponent(m.away)}&home_odds=${m.h}&draw_odds=${m.d}&away_odds=${m.a}`; const res = await fetch(url); const data = await res.json(); let isPlay = data.verdict && data.verdict.startsWith("RECOMMENDED PLAY"); let badgeClass = isPlay ? "badge-play" : "badge-noplay"; let outcomesHtml = ''; if (data.outcomes) { outcomesHtml = data.outcomes.map(o => `
• ${o.market}: Model ${o.model_prob_pct}% vs Mkt ${o.market_prob_pct}% | Edge: ${o.edge_pct}% | Kelly: ${o.suggested_stake_pct_bankroll || 0}%
` ).join(''); } html += `
${data.match || (m.home + ' vs ' + m.away)} ${data.safety_grade || 'ANALYZED'}
${outcomesHtml}
xG [H|A]: ${JSON.stringify(data.calculated_xG_detail || data.calculated_xG)} | Margin: ${data.market_margin_pct}%
`; } catch (err) { html += `
Error calculating ${m.home} vs ${m.away}
`; } } outputEl.innerHTML = html; }
